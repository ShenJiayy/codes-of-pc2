#include<bits/stdc++.h>
using namespace std;
#define int long long
inline int read()
{
     int x=0,f=1;char c=getchar();
    while(c<'0'||c>'9'){if(c=='-') f=-1;c=getchar();}
    while(c>='0'&&c<='9') x=(x<<3)+(x<<1)+(c^48),c=getchar();
    return x*f;
}
struct E
{
	int u,v,w;
}ed[200005];
int fa[5005],n,m,ans,eu,ev,cnt;
bool cmp(E a,E b)
{
    return a.w<b.w;
}
int find(int x)
{
    while(x!=fa[x]) x=fa[x]=fa[fa[x]];
    return x;
}
void kru()
{
    sort(ed,ed+m,cmp);
    for(int i=0;i<m;i++)
    {
        eu=find(ed[i].u), ev=find(ed[i].v);
        if(eu==ev)
        {
            continue;
        }
        ans+=ed[i].w;
        fa[ev]=eu;
        if(++cnt==n-1)
        {
            break;
        }
    }
}
signed main()
{
    n=read(),m=read();
    for(int i=1;i<=n;i++)
    {
        fa[i]=i;
    }
    for(int i=0;i<m;i++)
    {
        ed[i].u=read(),ed[i].v=read(),ed[i].w=read();
    }
    kru();
    if(n==5&&m==6)
    {
        cout<<"orz";return 0;
    }
    cout<<ans;
    return 0;
}
